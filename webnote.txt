ALTER PROCEDURE [dbo].[sp_OrderHeaders_Copy]
(
    @OldId          INT,
    @NewID          INT,
    @OrderId        NVARCHAR(100),
    @OrderNumber    NVARCHAR(100),
    @OrderName      NVARCHAR(255),
    @OrderNote      NVARCHAR(MAX),
    @CreatedBy      INT
)
AS
BEGIN

    INSERT INTO OrderHeaders
    SELECT
        @NewID,                 -- Id
        @OrderId,               -- OrderId

        @OrderNumber,						-- OrderNumber
        @OrderName,							-- OrderName

        GETDATE(),							-- CreatedDate
        NULL,										-- QuotedDate

        NULL,										-- InvoiceNumber
        NULL,										-- Collector

        0,											-- Amount
        NULL,   -- PackingNumber

END
