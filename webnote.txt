SELECT
	OrderDetails.Id,
	OrderDetails.HeaderId,
	OrderDetails.Charger AS [Motor Charger],
	OrderDetails.ExtensionCable AS [Extension Cable],
	OrderDetails.Supply AS [Neo Box],
	OrderDetails.SpringAssist AS [Spring Assist],
	OrderDetails.FrameColour,
	OrderDetails.AngleLength,
	OrderDetails.PlungerPin,
	OrderDetails.HingeQtyPerPanel,
	OrderDetails.PanelQtyWithHinge,
	OrderDetails.CustomHeaderLength,
	OrderDetails.TopTrack AS [Top Track],
	OrderDetails.BracketExtension AS [Extension Bracket],
	CASE
		WHEN OrderDetails.Sloping = 'Top' THEN 'Yes'
	END AS [Top Vertical Sloping],
	CASE
		WHEN OrderDetails.Sloping = 'Bottom' THEN 'Yes'
	END AS [Bottom Vertical Sloping],
	CASE
		WHEN OrderDetails.DoorCloser='Yes' AND Products.TubeType IN (23, 24) THEN 'Yes'
	END AS [Hinge Door Closer],
	
	CASE
		WHEN OrderDetails.DoorCloser='Yes' AND Products.TubeType IN (25, 26) THEN 'Yes'
	END AS [Sliding Door Closer],
	
	CASE WHEN Blinds.Name='Track Bi-fold' THEN 'Yes' END AS [Shutter Bi-fold],
	
	CASE
		WHEN OrderDetails.Buildout IN (
			'9.5mm Buildout',
			'25mm Buildout'
			)
		THEN 'Yes'
	END AS [Shutter Buildout],
	
	CASE
		WHEN
			OrderDetails.LayoutCodeCustom LIKE '%C%' OR
			OrderDetails.LayoutCodeCustom LIKE '%B%'
		THEN 'Yes'
	END AS [Shutter Corner Bay],
	
	CASE WHEN OrderDetails.TrackQty=1 THEN 'Yes' END AS [Sliding Single Track],
	CASE WHEN OrderDetails.TrackQty=2 THEN 'Yes' END AS [Sliding 2 Track],
	CASE WHEN OrderDetails.TrackQty=3 THEN 'Yes' END AS [Sliding 3 Track],
	
	CASE WHEN OrderDetails.SpecialShape='Yes' THEN 'Yes' END AS [Special Shape],
	
	CASE
		WHEN
			OrderDetails.SpecialShape='Yes' AND
			OrderDetails.TemplateProvided='Yes'
		THEN 'Yes'
	END AS [Template Shipping Fee],
	
	CASE
		WHEN OrderDetails.Painting = 'Yes' THEN 'Yes'
